import { Taxes } from "./irs";

export class Programmer implements OnError, OnPayday {
	private balance: number = 0;
	onError() {
		console.log("Goodbye, world!");
	}

	onPayday(money: number) {
		balance += (money - Taxes.IncomeTax);
	}
}
